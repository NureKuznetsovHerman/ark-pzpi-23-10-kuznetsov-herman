МЕТА РОБОТИ
Ознайомитися з основними методами рефакторингу коду, навчитися 
покращувати структуру програм без зміни їхньої зовнішньої поведінки. 
Засвоїти практичні прийоми інкапсуляції даних, обробки помилок через 
винятки та переходу від процедурного до об’єктно-орієнтованого підходу.
 
ХІД РОБОТИ
1. Метод Encapsulate Collection (Інкапсуляція колекцій)
Проблема:
У початковому коді вся робота з часом виконувалася прямо у функції main. 
Змінна структури tm використовувалась безпосередньо, а виклик localtime_s() 
здійснювався без жодного контролю доступу. 
Такий підхід ускладнює повторне використання коду, тестування і порушує 
принцип інкапсуляції, адже структура tm є відкритою для змін з будь-якої 
частини програми.
  
Рис. 1 – Код до рефакторінгу


Рефакторинг:
Було створено клас TimeWrapper, який інкапсулює структуру tm. Доступ до часу 
здійснюється лише через методи SetTime() і GetTime().
 
Рис. 2 – Код після рефакторінгу
Результат:
•	Дані приховані всередині класу. Покращена читабельність та повторне 
використання. Код став безпечнішим і зручнішим для тестування. Клас 
TimeWrapper можна легко застосувати у будь-якому іншому модулі, не 
дублюючи код.
2. Метод Replace Error Code with Exception (Замінити коди помилок на винятки)
Проблема:
Початковий код використовував коди повернення (return 1) для сигналізації про 
помилки. Такий підхід ускладнював підтримку й міг призвести до ігнорування 
помилок.
 
Рис. 3 – Код до рефакторінгу
Рефакторинг:
Усі критичні помилки тепер обробляються через винятки (throw runtime_error), а 
в main() додано блок try...catch.
 
Рис. 4 – Код після рефакторінгу
Результат:
Централізована обробка помилок. Код став гнучкішим і безпечнішим. Зменшено 
дублювання перевірок у функціях.
3. Метод Replace Delegation with Inheritance (Замінити делегування на 
наслідування)
Проблема:
Множення матриць реалізовувалося через сирі масиви int**, що створювало 
дублювання коду, ризик витоку пам’яті й складність у підтримці.
 
Рис. 5 – Код до рефакторінгу
Рефакторинг:
Було створено клас Matrix, який інкапсулює дані й операції над ними.
Для зберігання елементів використано std::vector, який автоматично очищує 
пам’ять.
 
Рис. 6 – Код після рефакторінгу
Результат:
Підвищена модульність і безпека пам’яті. Код став компактнішим і придатним до 
розширення. Зменшено дублювання процедурного коду.

розробника.

#include <stdio.h>
#include <time.h>
#include <windows.h>
#include <locale.h>

class TimeWrapper {
private:
    struct tm timeinfo{};
public:
    bool SetTime(time_t raw_time) {
        return localtime_s(&timeinfo, &raw_time) == 0;
    }
    struct tm GetTime() const { return timeinfo; }
};

int main() {
    setlocale(LC_ALL, "Ukrainian");
    SetConsoleOutputCP(1251);

    time_t max_32bit_time = 0x7FFFFFFF;
    TimeWrapper tw;
    if (!tw.SetTime(max_32bit_time)) {
        printf("Помилка перетворення часу\n");
        return 1;
    }

    struct tm t = tw.GetTime();
    printf("Остання дата для 32-бітного time_t: %02d.%02d.%d %02d:%02d:%02d\n",
           t.tm_mday, t.tm_mon + 1, t.tm_year + 1900,
           t.tm_hour, t.tm_min, t.tm_sec);
    return 0;
}

// Функції для вимірювання точності (залишаються незмінними)
double measure_time_precision() {
 vector<double> diffs;
 const int measurements = 100;
 for (int i = 0; i < measurements; i++) {
 time_t start = time(nullptr);
 time_t end;
 do {
 end = time(nullptr);
 } while (end == start);
 diffs.push_back(difftime(end, start));
 }
 return *min_element(diffs.begin(), diffs.end());
}
double measure_clock_precision() {
 vector<double> diffs;
 const int measurements = 100;
 for (int i = 0; i < measurements; i++) {
 clock_t start = clock();
 clock_t end;
 do {
 end = clock();
 } while (end == start);
 diffs.push_back((double)(end - start) / CLOCKS_PER_SEC *
1000.0);
 }
 return *min_element(diffs.begin(), diffs.end());
}
double measure_GetSystemTimeAsFileTime_precision() {
 vector<double> diffs;
 const int measurements = 100;
 for (int i = 0; i < measurements; i++) {
 FILETIME start, end;
 GetSystemTimeAsFileTime(&start);
 do {
 GetSystemTimeAsFileTime(&end);

 start.dwLowDateTime == end.dwLowDateTime);

 start_uli.LowPart = start.dwLowDateTime;
 start_uli.HighPart = start.dwHighDateTime;
 end_uli.LowPart = end.dwLowDateTime;
 end_uli.HighPart = end.dwHighDateTime;

double measure_QueryPerformanceCounter_precision() {

 LARGE_INTEGER frequency;
 QueryPerformanceFrequency(&frequency);
 for (int i = 0; i < measurements; i++) {
 LARGE_INTEGER start, end;
 QueryPerformanceCounter(&start);
 do {
 QueryPerformanceCounter(&end);
 } while (end.QuadPart == start.QuadPart);
 diffs.push_back((end.QuadPart - start.QuadPart) * 1000000.0 /
frequency.QuadPart);
 }
 return *min_element(diffs.begin(), diffs.end());
}
double measure_chrono_precision() {
 vector<double> diffs;
 const int measurements = 100;
 for (int i = 0; i < measurements; i++) {
 auto start = high_resolution_clock::now();
 auto end = high_resolution_clock::now();
 while (end == start) {
 end = high_resolution_clock::now();
 }
 auto duration = duration_cast<nanoseconds>(end - start);
 diffs.push_back(static_cast<double>(duration.count()));
 }
 return *min_element(diffs.begin(), diffs.end());
}

// Вимірювання за допомогою __rdtsc
double measure_array_sum_with_rdtsc() {
 const int SIZE = 1000;
 vector<int> array(SIZE, 1); // Масив з 1000 одиниць
 const int ITERATIONS = 10000; // Багато ітерацій для точності
 unsigned long long total_cycles = 0;
 for (int iter = 0; iter < ITERATIONS; iter++) {
 int sum = 0;
 unsigned long long start = __rdtsc();
 // Додавання елементів масиву
 for (int i = 0; i < SIZE; i++) {
 sum += array[i];
 }
 unsigned long long end = __rdtsc();
 total_cycles += (end - start);
 // Використовуємо sum щоб уникнути оптимізації
 if (sum != SIZE) {
 cout << "Помилка в обчисленнях!" << endl;
 }
 }
 return static_cast<double>(total_cycles) / ITERATIONS;
}
// Вимірювання за допомогою QueryPerformanceCounter
double measure_array_sum_with_qpc() {
 const int SIZE = 1000;
 vector<int> array(SIZE, 1);
 const int ITERATIONS = 10000;
 LARGE_INTEGER frequency;
 QueryPerformanceFrequency(&frequency);
 double total_time = 0.0;
 for (int iter = 0; iter < ITERATIONS; iter++) {
 int sum = 0;
 LARGE_INTEGER start, end;
 QueryPerformanceCounter(&start);
 // Додавання елементів масиву
 for (int i = 0; i < SIZE; i++) {
 sum += array[i];
 }
 QueryPerformanceCounter(&end);
 double iteration_time = (end.QuadPart - start.QuadPart) *
1000000.0 / frequency.QuadPart;
 total_time += iteration_time;
 if (sum != SIZE) {
 cout << "Помилка в обчисленнях!" << endl;
 }
 }
 return total_time / ITERATIONS;
}

// Вимірювання з __rdtsc з обробкою виключень
double measure_array_sum_with_rdtsc() {
    const int SIZE = 1000;
    vector<int> array(SIZE, 1);
    const int ITERATIONS = 10000;
    unsigned long long total_cycles = 0;

    for (int iter = 0; iter < ITERATIONS; iter++) {
        int sum = 0;
        unsigned long long start = __rdtsc();
        for (int i = 0; i < SIZE; i++)
            sum += array[i];
        unsigned long long end = __rdtsc();

        if (sum != SIZE)
            throw runtime_error("Помилка в обчисленнях при RDTSC!");

        total_cycles += (end - start);
    }
    return static_cast<double>(total_cycles) / ITERATIONS;
}

// Вимірювання з QueryPerformanceCounter з обробкою виключень
double measure_array_sum_with_qpc() {
    const int SIZE = 1000;
    vector<int> array(SIZE, 1);
    const int ITERATIONS = 10000;
    LARGE_INTEGER frequency;
    if (!QueryPerformanceFrequency(&frequency))
        throw runtime_error("Помилка: неможливо отримати частоту таймера QPC!");

    double total_time = 0.0;
    for (int iter = 0; iter < ITERATIONS; iter++) {
        int sum = 0;
        LARGE_INTEGER start, end;
        QueryPerformanceCounter(&start);
        for (int i = 0; i < SIZE; i++)
            sum += array[i];
        QueryPerformanceCounter(&end);

        if (sum != SIZE)
            throw runtime_error("Помилка в обчисленнях при QPC!");

        double iteration_time = (end.QuadPart - start.QuadPart) * 1000000.0 / 
frequency.QuadPart;
        total_time += iteration_time;
    }
    return total_time / ITERATIONS;
}

// Вимірювання з std::chrono з обробкою виключень
double measure_array_sum_with_chrono() {
    const int SIZE = 1000;
    vector<int> array(SIZE, 1);
    const int ITERATIONS = 10000;
    double total_time = 0.0;

    for (int iter = 0; iter < ITERATIONS; iter++) {
        int sum = 0;
        auto start = high_resolution_clock::now();
        for (int i = 0; i < SIZE; i++)
            sum += array[i];
        auto end = high_resolution_clock::now();

        if (sum != SIZE)
            throw runtime_error("Помилка в обчисленнях при chrono!");

        auto duration = duration_cast<nanoseconds>(end - start);
        total_time += duration.count() / 1000.0;
    }
    return total_time / ITERATIONS;
}
